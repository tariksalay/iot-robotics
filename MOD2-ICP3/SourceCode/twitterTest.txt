#!/usr/bin/env python3
"""Historical Twitter/Twython exercise; preview by default, post only with --send."""
import argparse
import os


def main(argv=None):
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument('--text', default='Coursework API demonstration')
    parser.add_argument('--send', action='store_true', help='Actually submit the configured text')
    args = parser.parse_args(argv)
    if not args.send:
        print('Preview only:', args.text)
        return
    keys = ['TWITTER_API_KEY', 'TWITTER_API_SECRET', 'TWITTER_ACCESS_TOKEN', 'TWITTER_ACCESS_TOKEN_SECRET']
    if any(not os.getenv(key) for key in keys):
        parser.error('Set all four TWITTER_* credential environment variables before sending.')
    from twython import Twython
    api = Twython(*(os.environ[key] for key in keys))
    api.update_status(status=args.text)
    print('Submitted the configured text.')


if __name__ == '__main__':
    main()
